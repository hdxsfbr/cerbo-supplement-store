export function platformFeeCents(totalCents) {
  const scaled = totalCents * 75;
  return Math.floor((scaled + 5000) / 10000); // half-up
}

export function computeSplit(items) {
  total  = Σ quantity × unitPriceCents
  cogs   = Σ quantity × unitCogsCents   // snapshot
  fee    = platformFeeCents(total)
  margin = total − cogs − fee       // residual ⇒ exact
  memos  = { platform_fee: "75 bps of 6600¢ = 49.5¢ → 50¢ (half-up)", … }
}
